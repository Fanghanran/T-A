# 日志系统与异常系统设计文档

> 日期：2026-08-25
> 状态：待实现

---

## 1. 现状问题

**后端**：
- 全项目依赖 `console.log/warn/error`，无结构化输出
- `dbg()` 开关只在 `index.js`，`chunker.js`/`embed.js`/`llm.js`/`vectorStore.js` 各自散落的 console 无法统一控制
- 无请求追踪（一次请求跨越多个模块，无法串联日志）
- 错误只 `console.error`，无请求上下文（谁、什么接口、什么参数、什么阶段）
- 无错误日志持久化，服务重启后日志消失

**前端**：
- 无全局错误捕获（`window.onerror`/`unhandledrejection` 未注册）
- 每个 hook 各自 catch，没有统一的错误上报/展示机制
- 网络错误、JSON 解析失败、API 返回 5xx 没有任何兜底

---

## 2. 设计目标

1. **结构化日志**：每次 log 包含时间、级别、模块、请求ID、消息
2. **请求追踪**：一次请求全链路可追溯（一个 requestId 贯穿所有日志）
3. **统一异常处理**：所有错误统一收集，不遗漏
4. **前端兜底**：全局 error boundary + 网络错误提示
5. **零侵入改造**：用 logger 包装替换 console，不改变业务逻辑

---

## 3. 后端设计

### 3.1 依赖

```bash
npm install pino pino-http
```

- `pino`：结构化 JSON 日志引擎，性能极快（比 winston/bunyan 快 3-5x）
- `pino-http`：Express 请求日志中间件，自动生成 requestId

### 3.2 模块：`server/lib/logger.js`

```js
import pino from 'pino'
import process from 'node:process'

// 日志文件路径：data/logs/app.log
const LOG_FILE = process.env.LOG_FILE || 'data/logs/app.log'

// 生产：写入文件（带滚动）；开发：输出到 stdout
const transport = process.env.NODE_ENV === 'production'
  ? {
      target: 'pino/file',
      options: { destination: LOG_FILE, transport: { target: 'pino-pretty' } }
    }
  : undefined

export const logger = pino({
  level: process.env.LOG_LEVEL || 'info',
  timestamp: pino.stdTimeFunctions.isoTime,
  base: { app: 'interview-agent' },
  ...(transport ? { transport } : {}),
})

/** 带上下文的前缀 logger（用于模块注入） */
export function childLogger(moduleName) {
  return logger.child({ module: moduleName })
}

/** 从 request 中提取 requestId，用于跨模块关联 */
export function logRequestInfo(req, res, next) {
  // pino-http 自动注入 req.log
  next()
}
```

### 3.3 模块：`server/lib/errors.js` — 统一异常体系

```js
/**
 * 业务异常基类
 * - 区分"客户端错误"（4xx）和"服务端错误"（5xx）
 * - 携带 code 字段便于前端展示友好提示
 */
export class AppError extends Error {
  constructor(message, { status = 500, code = 'INTERNAL_ERROR', cause } = {}) {
    super(message)
    this.name = 'AppError'
    this.status = status
    this.code = code
    this.cause = cause
  }
}

/** 分类 */
export class NotFoundError extends AppError {
  constructor(message) { super(message, { status: 404, code: 'NOT_FOUND' }) }
}
export class BadRequestError extends AppError {
  constructor(message) { super(message, { status: 400, code: 'BAD_REQUEST' }) }
}
export class ServiceUnavailableError extends AppError {
  constructor(message) { super(message, { status: 503, code: 'SERVICE_UNAVAILABLE' }) }
}

/**
 * 统一错误处理中间件（Express）
 * - 区分 AppError（已知业务错误）和普通 Error（未知异常）
 * - 记录结构化日志（请求方法、路径、status、error code、stack）
 * - 统一响应格式
 */
export function errorHandler(err, req, res, next) {
  const isAppError = err instanceof AppError
  const status = isAppError ? err.status : 500
  const code = isAppError ? err.code : 'INTERNAL_ERROR'

  // 结构化日志
  req.log?.error?.({
    msg: err.message,
    code,
    status,
    stack: err.stack,
    module: err.module,
  }, err.message)

  // 或 fallback：logger.error({ code, status, msg: err.message, stack: err.stack })

  res.status(status).json({
    success: false,
    code,
    message: isAppError ? err.message : (
      process.env.NODE_ENV === 'production'
        ? '内部错误，请稍后重试'
        : err.message
    ),
  })
}
```

### 3.4 模块：`server/lib/requestTrace.js` — 请求追踪

```js
import { randomUUID } from 'node:crypto'

/**
 * 为每次请求生成唯一 requestId，注入到 res.locals 和 logger context
 * 配合 pino-http 使用：每个请求自动带 requestId
 */
export function requestTrace() {
  return (req, res, next) => {
    const requestId = req.headers['x-request-id'] || randomUUID()
    req.requestId = requestId
    res.locals.requestId = requestId
    res.setHeader('x-request-id', requestId)
    // 把 requestId 注入到本次请求的 logger context
    req.log = req.log?.child?.({ requestId }) || logger.child({ requestId })
    next()
  }
}
```

## 4. 前端设计

### 4.1 模块：`src/lib/logger.js`

```js
/**
 * 前端轻量日志工具
 * - DEBUG_LOG=true 时输出到 console
 * - 生产环境只记录 error，不刷屏
 * - 支持命名空间隔离（chat / knowledge / session）
 */
const DEBUG = typeof import.meta.env?.DEBUG_LOG !== 'undefined'
  ? import.meta.env.DEBUG_LOG
  : /^(1|true|yes|on)$/i.test(String(localStorage.getItem('DEBUG_LOG') ?? ''))

const prefixes = {
  chat: '[Chat]',
  knowledge: '[Knowledge]',
  session: '[Session]',
  api: '[API]',
  error: '[Error]',
}

function format(namespace, args) {
  const prefix = prefixes[namespace] || `[${namespace}]`
  return [prefix, ...args].join(' ')
}

export const logger = {
  debug: (...args) => { if (DEBUG) console.debug(...args) },
  info:  (...args) => { if (DEBUG) console.info(...args) },
  warn:  (...args) => console.warn(...args),   // warn 始终输出
  error: (...args) => console.error(...args),  // error 始终输出
}

export function child(namespace) {
  return {
    debug: (...a) => logger.debug(format(namespace, a)),
    info:  (...a) => logger.info(format(namespace, a)),
    warn:  (...a) => logger.warn(format(namespace, a)),
    error: (...a) => logger.error(format(namespace, a)),
  }
}
```

### 4.2 模块：`src/lib/api.js` — 统一 API 客户端

```js
import { child } from './logger'

const log = child('api')

/**
 * 统一 fetch 封装：
 * - 自动附加 requestId（x-request-id header）
 * - 统一错误处理：解析错误体 → 抛 AppError → 触发全局错误边界
 * - 请求/响应日志（DEBUG 模式下）
 */
let _requestCount = 0

export async function request(url, options = {}) {
  const id = ++_requestCount
  const requestId = `req-${Date.now()}-${id}`

  log.debug(`#${id} ${options.method ?? 'GET'} ${url}`, { requestId })

  const res = await fetch(url, {
    ...options,
    headers: {
      'x-request-id': requestId,
      ...(options.headers ?? {}),
    },
  })

  const contentType = res.headers.get('content-type') || ''

  // 流式响应直接返回（不解析 JSON）
  if (contentType.includes('text/event-stream') || contentType.includes('application/octet-stream')) {
    return res
  }

  // 204 No Content
  if (res.status === 204) return null

  let data
  try {
    data = await res.json()
  } catch {
    throw new AppError(`响应解析失败 (${res.status})`, { status: res.status, requestId })
  }

  if (!res.ok) {
    const message = data?.message ?? `请求失败 (${res.status})`
    log.error(`#${id} 失败 ${res.status}: ${message}`, { requestId, code: data?.code })
    throw new AppError(message, { status: res.status, code: data?.code, requestId })
  }

  log.debug(`#${id} 成功 ${res.status}`, { requestId })
  return data
}

export class AppError extends Error {
  constructor(message, { status, code, requestId } = {}) {
    super(message)
    this.name = 'AppError'
    this.status = status
    this.code = code
    this.requestId = requestId
  }
}
```

### 4.3 模块：`src/components/ErrorBoundary.jsx` — 全局错误边界

```jsx
import { child } from '@/lib/logger'
const log = child('error')

export class ErrorBoundary extends React.Component {
  state = { hasError: false, error: null }

  static getDerivedStateFromError(error) {
    return { hasError: true, error }
  }

  componentDidCatch(error, info) {
    log.error('React 组件异常', { error: error.message, stack: info.componentStack })
  }

  render() {
    if (this.state.hasError) {
      return (
        <div className="flex h-dvh items-center justify-center gap-3 text-sm text-destructive">
          <AlertCircle className="h-5 w-5 shrink-0" />
          <div>
            <p className="font-medium">应用遇到异常</p>
            <p className="text-xs text-muted-foreground mt-1">
              {this.state.error?.message ?? '未知错误'}
            </p>
            <Button
              size="sm"
              variant="outline"
              className="mt-3"
              onClick={() => { this.setState({ hasError: false, error: null }); window.location.reload() }}
            >
              刷新页面
            </Button>
          </div>
        </div>
      )
    }
    return this.props.children
  }
}
```

### 4.4 模块：`src/hooks/useGlobalError.ts` — 全局错误监听

```js
import { useEffect } from 'react'
import { child } from '@/lib/logger'
const log = child('error')

/**
 * 注册 window 级错误监听：
 * - unhandledrejection：Promise 未 catch
 * - error：脚本/资源加载错误
 * 统一上报到 logger，不干扰业务逻辑
 */
export function useGlobalError() {
  useEffect(() => {
    const handler = (e) => {
      const reason = e.reason ?? e.message
      log.error('未处理异常', {
        type: e.type,
        message: reason?.toString?.() ?? String(reason),
        filename: e.filename,
        lineno: e.lineno,
        colno: e.colno,
      })
    }
    window.addEventListener('error', handler)
    window.addEventListener('unhandledrejection', handler)
    return () => {
      window.removeEventListener('error', handler)
      window.removeEventListener('unhandledrejection', handler)
    }
  }, [])
}
```

## 5. 日志级别说明

| 级别 | 用途 | 示例 |
|------|------|------|
| `info` | 启动、健康检查、关键业务流程 | "服务已启动"、"会话创建" |
| `warn` | 降级、非致命失败、配置缺失 | "Embedding 降级为 hash"、"LLM 不可用" |
| `error` | 未预期的错误、500 响应 | "SQLite 写入失败"、"Vectra 连接超时" |
| `debug` | 请求详情、协议采样、内部状态 | 受 DEBUG_LOG 控制，生产不输出 |
