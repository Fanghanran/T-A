/**
 * 文件条目状态 → 右侧状态文案
 * @param {{ status: string, chunkCount?: number, stage?: string }} f
 */
export function entryStatusText(f) {
  switch (f.status) {
    case 'preparing':
      return '解析切片中…'
    case 'ready':
      return f.chunkCount != null ? `就绪 · ${f.chunkCount} 块` : '就绪'
    case 'committing':
      return f.stage || '排队中…'
    default:
      return ''
  }
}
